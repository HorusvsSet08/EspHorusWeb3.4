const char* mqtt_topic = "horus/estacion/h001/telemetry";

const char* thingspeak_api_key = "TU_THINGSPEAK_WRITE_KEY"; // opcional

WiFiClient espClient;
PubSubClient mqttClient(espClient);
HardwareSerial PMserial(2); // Serial2 -> RX2/TX2 (GPIO16,GPIO17)

Adafruit_BME280 bme; // I2C
// RTC si procede

const int SD_CS = 5; // pin CS SD
const int PIN_ANEMO = 4; // pulso anemómetro
volatile unsigned long anemoPulses = 0;
unsigned long lastAnemoCalc = 0;
float wind_speed = 0.0;

void IRAM_ATTR onAnemoPulse() {
  anemoPulses++;
}

void setup() {
  Serial.begin(115200);
  Wire.begin(21,22); // SDA, SCL

  // BME init
  if (!bme.begin(0x76)) {
    Serial.println("No BME280 detectado en 0x76");
  }

  // PM sensor init (PMS -> Serial2)
  PMserial.begin(9600, SERIAL_8N1, 16, 17); // RX2=16, TX2=17

  // SD init
  if (!SD.begin(SD_CS)) {
    Serial.println("Fallo inicializando SD");
  } else {
    Serial.println("SD inicializada");
  }

  // WiFi y MQTT
  WiFi.begin(ssid, password);
  Serial.print("Conectando WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500); Serial.print(".");
  }
  Serial.println(" conectado");
  mqttClient.setServer(mqtt_server, mqtt_port);

  // Interrupción anemómetro
  pinMode(PIN_ANEMO, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(PIN_ANEMO), onAnemoPulse, FALLING);

  lastAnemoCalc = millis();
}

  int pm10 = -1;
  if (PMserial.available()) {
    // parsear paquete PMS --- aquí debes usar parseador específico
    // ejemplo simplificado pseudo:
    // pm25 = ...; pm10 = ...;
  }

  // Anemómetro — calcular cada periodo (ej. 5s)
  if (millis() - lastAnemoCalc >= 5000) {
    noInterrupts();
    unsigned long pulses = anemoPulses;
    anemoPulses = 0;
    interrupts();
    // Ejemplo: convertir pulses en m/s según tu anemómetro
    // supongamos 1 pulse/s = 1.2 m/s (depende del sensor)
    wind_speed = (pulses / 5.0) * 1.2;
    lastAnemoCalc = millis();

  // JSON payload

  payload += "\"ts\":\"" + getISOtime() + "\",";
  payload += "\"id\":\"h001\",";
  payload += "\"temp\":" + String(temp,1) + ",";
  payload += "\"hum\":" + String(hum,1) + ",";
  payload += "\"pres\":" + String(pres,1) + ",";
  payload += "\"alt\":" + String(altitude,1) + ",";
  payload += "\"wind_speed\":" + String(wind_speed,2) + ",";
  payload += "\"rain_mm\":" + String(rain_mm,2) + ",";
  payload += "\"pm25\":" + String(pm25) + ",";
  payload += "\"pm10\":" + String(pm10) + ",";
  payload += "\"gas_raw\":" + String(gasRaw);
  payload += "}";

  // Publicar MQTT
  mqttClient.publish(mqtt_topic, payload.c_str());

  // Subir a ThingSpeak (ejemplo mínimo, usa HTTPClient)

      f.close();

Notas específicas y adaptaciones para tus módulos

  • PMS5003/PMS6003/etc. envía datos por UART en un paquete binario; usa Serial2 y una librería de parsing (p. ej. PMS.h) para extraer PM1.0/2.5/10.
  • AS5600 (veleta) es un sensor magnético I²C que devuelve ángulo en 0–4095; convierte a grados (ángulo * 360 / 4096) y mapea a dirección meteorológica (0 = N, etc.).
  • Anemómetro Hall requiere contar pulsos. Calibra pulsos→m/s con la curva del fabricante o comparando con anemómetro de referencia.
  • MQ necesita tiempo de calentamiento (24–48 h) y lectura de referencia; su valor ADC es aproximado y requiere calibración y/o circuitería adicional para RL.
  • Rain module algunos dan salida digital (contacto) y otros analógica; el pluviómetro de taza basculante normalmente genera impulsos (cada basculada = volumen X → mm). Tu módulo "Snow Raindrops Detection" puede dar señal digital/analógica: verifica si es "tipping bucket" o resistivo.
  • SD usa nivel lógico 3.3V y la interfaz SPI propuesta (CS GPIO5). Evita conflictos con otros periféricos SPI.

Calibración rápida (resumen)

  • BME280: comparar con estación local o termómetro calibrado; aplicar offset de +/−0.5°C si es constante.
  • PMS: dejar 24–48 h de estabilización; compara con estación de referencia y calcula offset multiplicativo si hace falta.
  • Anemómetro: calibrar en campo con tabla de fabricante o comparador.
  • Pluviómetro: cada pulso = X mm (según modelo). Mide manualmente 1 L en contenedor y cuenta pulsos para calcular coeficiente.
  • MQ: calibrar con aire limpio y luego con gas de referencia si es posible.

Preguntas frecuentes específicas (respuestas rápidas)

¿Dónde conecto la SD y la RTC si ambas usan SPI / I²C?

La RTC y los sensores I²C comparten I²C (GPIO21/22). La SD usa SPI (GPIO23/19/18/5 para MOSI/MISO/SCK/CS). Usa CS dedicado y evita que otros usos SPI interfieran.

¿Cómo se comunica con el Arduino Nano?

Si el Nano tiene Bluetooth (p.ej. módulo HC-05), el ESP32 puede usar Bluetooth nativo para emparejar y enviar comandos; alternativa: UART físico entre ESP32 y Nano (nivel TTL) si están en la misma caja.

¿Qué pasa si el broker HiveMQ cae?

HT: HiveMQ público puede ser inestable; el firmware intenta reconectar. Guarda en SD si no hay conexión y sincroniza cuando vuelva.
